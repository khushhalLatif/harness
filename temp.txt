import json
import yaml
import requests

from harness_config import load_config, require_harness

config = load_config()
harness = require_harness(config)

# The Harness SERVICE identifier — not the artifact source id ("uivaultportal")
# or the manifest id ("mnportal") from your earlier screenshots. Grab it from
# the "Service Identifier" column in the master Excel Script 1 already produced.
SERVICE_IDENTIFIER = "PUT_ONE_SERVICE_IDENTIFIER_HERE"

headers = {"x-api-key": harness["api_key"]}
params = {
    "accountIdentifier": harness["account_id"],
    "orgIdentifier": harness["org_id"],
    "projectIdentifier": harness["project_id"],
}

url = f"{harness['base_url']}/ng/api/servicesV2/{SERVICE_IDENTIFIER}"
resp = requests.get(url, headers=headers, params=params)
resp.raise_for_status()

parsed = yaml.safe_load(resp.json()["data"]["service"]["yaml"])
spec = parsed["service"]["serviceDefinition"]["spec"]

print("=== Top-level keys in spec ===")
print(list(spec.keys()))

print("\n=== Raw 'artifacts' block ===")
print(json.dumps(spec.get("artifacts", "NO ARTIFACTS KEY AT ALL"), indent=2))

print("\n=== Raw 'manifests' block (for comparison) ===")
print(json.dumps(spec.get("manifests", "NO MANIFESTS KEY"), indent=2))
