import requests

BASE_URL = "https://gitlab.onefiserv.net"
TOKEN = "YOUR_REAL_TOKEN"
headers = {"PRIVATE-TOKEN": TOKEN}

# Test 1: is the token valid at all, no group needed?
r1 = requests.get(f"{BASE_URL}/api/v4/user", headers=headers)
print("User call:", r1.status_code, r1.text[:400])

# Test 2: can it see the specific group?
group_path = "na/gfs/prepaid/moneynetwork/mn-2.0"
encoded = requests.utils.quote(group_path, safe="")
r2 = requests.get(f"{BASE_URL}/api/v4/groups/{encoded}", headers=headers)
print("Group call:", r2.status_code, r2.text[:400])
